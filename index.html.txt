    <script>
        // CẤU HÌNH LIÊN KẾT: Đã tự động điền link Web App hoạt động của bạn
        const API_URL = "https://google.com";
        
        // THÔNG TIN TÀI KHOẢN NGÂN HÀNG ĐỂ TẠO MÃ QR TỰ ĐỘNG
        const BANK_ID = "vietcombank"; // Thay bằng mã ngân hàng của bạn viết thường (ví dụ: mbbank, bidv, techcombank...)
        const ACCOUNT_NO = "0000000000000"; // Thay bằng Số tài khoản của bạn
        const ACCOUNT_NAME = "NGUYEN VAN A"; // Thay bằng Tên chủ tài khoản không dấu (in hoa)

        let currentUser = localStorage.getItem('quiz_user') || null;
        let currentXu = parseInt(localStorage.getItem('quiz_xu')) || 0;
        let allQuestions = [];
        let filteredQuestions = [];
        let currentQIndex = 0;
        let userScore = 0;
        let currentBoDe = "";
        let isAnswered = false;

        // Tính năng bảo mật nâng cao: Khóa chuột phải và các tổ hợp phím copy cơ bản
        document.addEventListener('keydown', function(e) {
            if (e.ctrlKey && (e.key === 'c' || e.key === 'v' || e.key === 'u' || e.key === 'a' || e.key === 's' || e.key === 'p')) {
                alert("Hành động sao chép bị nghiêm cấm trên hệ thống!");
                e.preventDefault();
                return false;
            }
            if (e.key === "F12" || (e.ctrlKey && e.shiftKey && e.key === "I")) {
                e.preventDefault();
                return false;
            }
        });

        function showScreen(screenId) {
            const screens = ['auth-screen', 'subject-screen', 'bode-screen', 'payment-screen', 'quiz-screen', 'result-screen'];
            screens.forEach(id => {
                const el = document.getElementById(id);
                if(el) el.classList.add('hidden');
            });
            const targetEl = document.getElementById(screenId);
            if(targetEl) targetEl.classList.remove('hidden');
        }

        function loadAllQuestions() {
            fetch(`${API_URL}?action=readQuestions`)
                .then(res => res.json())
                .then(data => { allQuestions = data; })
.catch(err => console.error("Lỗi đồng bộ câu hỏi:", err));
}
function handleAuth(type) {
const user = document.getElementById('username').value.trim();
const pass = document.getElementById('password').value.trim();
if(!user || !pass) return alert("Vui lòng nhập đầy đủ thông tin!");
fetch(${API_URL}?action=${type}&username=${encodeURIComponent(user)}&password=${encodeURIComponent(pass)})
.then(res => res.json())
.then(res => {
if(type === 'login' && res.result === 'success') {
currentUser = user;
currentXu = res.xu;
localStorage.setItem('quiz_user', user);
localStorage.setItem('quiz_xu', res.xu);
updateUserBar();
renderSubjects();
showScreen('subject-screen');
} else if(type === 'register' && res.result === 'success') {
alert("Đăng ký thành công! Hãy bấm đăng nhập.");
} else if(res.result === 'exists') {
alert("Tài khoản này đã tồn tại!");
} else {
alert("Sai tên đăng nhập hoặc mật khẩu!");
}
})
.catch(err => alert("Lỗi kết nối máy chủ dữ liệu!"));
}
function logout() {
localStorage.clear();
currentUser = null;
currentXu = 0;
updateUserBar();
showScreen('auth-screen');
}
function updateUserBar() {
if(currentUser) {
document.getElementById('user-display').innerText = Học viên: ${currentUser};
document.getElementById('xu-display').innerText = Xu: ${currentXu};
document.getElementById('logout-btn').classList.remove('hidden');
} else {
document.getElementById('user-display').innerText = Chưa đăng nhập;
document.getElementById('xu-display').innerText = Xu: 0;
document.getElementById('logout-btn').classList.add('hidden');
}
}
function renderSubjects() {
const listContainer = document.getElementById('subject-list');
listContainer.innerHTML = "";
const uniqueSubjects = [...new Set(allQuestions.map(item => item.mon))];
uniqueSubjects.forEach(mon => {
if(!mon) return;
const div = document.createElement('div');
div.className = 'grid-item';
div.innerText = mon;
div.onclick = () => renderBoDe(mon);
listContainer.appendChild(div);
});
}
function renderBoDe(subjectName) {
document.getElementById('selected-subject-title').innerText = Bộ Đề Môn: ${subjectName};
const listContainer = document.getElementById('bode-list');
listContainer.innerHTML = "";
const subQuestions = allQuestions.filter(q => q.mon === subjectName);
const uniqueBoDe = [...new Set(subQuestions.map(item => item.bode))];
uniqueBoDe.forEach(bode => {
if(!bode) return;
const div = document.createElement('div');
div.className = 'grid-item';
div.innerText = bode.replace(/_/g, ' ');
div.onclick = () => selectBoDe(bode);
listContainer.appendChild(div);
});
showScreen('bode-screen');
}
function selectBoDe(bodeName) {
currentBoDe = bodeName;
fetch(API_URL, {
method: "POST",
body: JSON.stringify({
action: "checkAndDeduct",
username: currentUser,
bode: currentBoDe
})
})
.then(res => res.json())
.then(res => {
if(res.status === "allow") {
if(res.message) alert(res.message);
if(res.newXu !== undefined) {
currentXu = res.newXu;
localStorage.setItem('quiz_xu', currentXu);
updateUserBar();
}
startQuiz(bodeName);
} else if(res.status === "block" && res.reason === "no_xu") {
document.getElementById('qr-content').innerText = NAPXU ${currentUser};
const qrUrl = https://vietqr.io{BANK_ID}-${ACCOUNT_NO}-print.png?amount=10000&addInfo=NAPXU%20${encodeURIComponent(currentUser)}&accountName=${encodeURIComponent(ACCOUNT_NAME)};
document.getElementById('vietqr-img').src = qrUrl;
showScreen('payment-screen');
}
})
.catch(err => alert("Không thể kiểm tra số dư xu lúc này!"));
}
function shuffle(array) {
for (let i = array.length - 1; i > 0; i--) {
const j = Math.floor(Math.random() * (i + 1));
[array[i], array[j]] = [array[j], array[i]];
}
return array;
}
function startQuiz(bodeName) {
let baseQuestions = allQuestions.filter(q => q.bode === bodeName);
filteredQuestions = shuffle([...baseQuestions]);
currentQIndex = 0;
userScore = 0;
showScreen('quiz-screen');
loadQuestion();
}
function loadQuestion() {
isAnswered = false;
document.getElementById('next-btn').style.display = 'none';
let currentQ = filteredQuestions[currentQIndex];
document.getElementById('quiz-progress').innerText = Câu ${currentQIndex + 1}/${filteredQuestions.length};
document.getElementById('quiz-question').innerText = currentQ.q;
const optionsBox = document.getElementById('quiz-options');
optionsBox.innerHTML = "";
const labels = ['A', 'B', 'C', 'D'];
currentQ.a.forEach((ans, index) => {
const div = document.createElement('div');
div.className = 'option-item';
div.innerText = ${labels[index]}. ${ans};
div.onclick = () => checkUserAnswer(div, labels[index], currentQ.correct);
optionsBox.appendChild(div);
});
}
function checkUserAnswer(element, selectedLetter, correctLetter) {
if(isAnswered) return;
isAnswered = true;
const options = document.querySelectorAll('.option-item');
const labels = ['A', 'B', 'C', 'D'];
let correctIndex = labels.indexOf(correctLetter);
if(selectedLetter === correctLetter) {
element.classList.add('correct');
userScore++;
} else {
element.classList.add('wrong');
if(correctIndex !== -1) {
options[correctIndex].classList.add('correct');
}
}
document.getElementById('next-btn').style.display = 'block';
}
document.getElementById('next-btn').onclick = function() {
currentQIndex++;
if(currentQIndex < filteredQuestions.length) {
loadQuestion();
} else {
document.getElementById('result-text').innerText = Bạn làm đúng ${userScore} / ${filteredQuestions.length} câu!;
showScreen('result-screen');
}
};
if(currentUser) {
updateUserBar();
showScreen('subject-screen');
}
loadAllQuestions();
